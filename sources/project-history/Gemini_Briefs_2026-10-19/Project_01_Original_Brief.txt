      I = I + 1
      GOTO 10

      STOP
      END

      IMPLICIT DOUBLE PRECISION (A-H,O-Z)
      COMMON /PARAM/ GM, DT, NSTEPS
      R2 = PX*PX + PY*PY
      R3 = R2 * DSQRT(R2)
      AX = -GM * PX / R3
      AY = -GM * PY / R3
      RETURN
      END

      COMMON /PARAM/ GM, DT, NSTEPS
      COMMON /STATE/ X, Y, VX, VY

      K1X = VX
      K1Y = VY
      K1VX = AX1
      K1VY = AY1

pub struct SimConfig {
    pub gm: f64,
    pub dt: f64,
    pub nsteps: usize,
}

#[derive(Debug, Clone, Copy)]
pub struct OrbitalState {
    pub x: f64,
    pub y: f64,
    pub vx: f64,
    pub vy: f64,
}

impl OrbitalState {
    #[inline]
    fn accel(px: f64, py: f64, gm: f64) -> (f64, f64) {
        let r2 = px * px + py * py;
        let r3 = r2 * r2.sqrt();
        (-gm * px / r3, -gm * py / r3)
    }

    pub fn rk4_step(&mut self, cfg: &SimConfig) {
        let dt = cfg.dt;
        let (ax1, ay1) = Self::accel(self.x, self.y, cfg.gm);
        let (k1x, k1y, k1vx, k1vy) = (self.vx, self.vy, ax1, ay1);

        let (ax2, ay2) = Self::accel(
            self.x + 0.5 * dt * k1x,
            self.y + 0.5 * dt * k1y,
            cfg.gm,
        );
        let (k2x, k2y, k2vx, k2vy) = (
            self.vx + 0.5 * dt * k1vx,
            self.vy + 0.5 * dt * k1vy,
            ax2,
            ay2,
        );

        let (ax3, ay3) = Self::accel(
            self.x + 0.5 * dt * k2x,
            self.y + 0.5 * dt * k2y,
            cfg.gm,
        );
        let (k3x, k3y, k3vx, k3vy) = (
            self.vx + 0.5 * dt * k2vx,
            self.vy + 0.5 * dt * k2vy,
            ax3,
            ay3,
        );

        let (ax4, ay4) = Self::accel(self.x + dt * k3x, self.y + dt * k3y, cfg.gm);
        let (k4x, k4y, k4vx, k4vy) = (self.vx + dt * k3vx, self.vy + dt * k3vy, ax4, ay4);

        let scale = dt / 6.0;
        self.x += scale * (k1x + 2.0 * k2x + 2.0 * k3x + k4x);
        self.y += scale * (k1y + 2.0 * k2y + 2.0 * k3y + k4y);
        self.vx += scale * (k1vx + 2.0 * k2vx + 2.0 * k3vx + k4vx);
        self.vy += scale * (k1vy + 2.0 * k2vy + 2.0 * k3vy + k4vy);
    }
}

fn main() {
    let cfg = SimConfig {
        gm: 3.986004418e14,

}

#include <iostream>
#include <utility>

struct SimConfig {
    double gm{3.986004418e14};
    double dt{1.0};
    int nsteps{1000};
};

struct OrbitalState {
    double x{6.778137e6};
    double y{0.0};
    double vx{0.0};
    double vy{7.6686e3};

    static constexpr std::pair<double, double> accel(double px, double py, double gm) noexcept {
        const double r2 = px * px + py * py;
        const double r3 = r2 * std::sqrt(r2);
        return {-gm * px / r3, -gm * py / r3};
    }

    void rk4_step(const SimConfig& cfg) noexcept {
        const double dt = cfg.dt;
        const auto [ax1, ay1] = accel(x, y, cfg.gm);
        const double k1x = vx, k1y = vy, k1vx = ax1, k1vy = ay1;

        const auto [ax2, ay2] = accel(x + 0.5 * dt * k1x, y + 0.5 * dt * k1y, cfg.gm);
        const double k2x = vx + 0.5 * dt * k1vx, k2y = vy + 0.5 * dt * k1vy;
        const double k2vx = ax2, k2vy = ay2;

        const auto [ax3, ay3] = accel(x + 0.5 * dt * k2x, y + 0.5 * dt * k2y, cfg.gm);
        const double k3x = vx + 0.5 * dt * k2vx, k3y = vy + 0.5 * dt * k2vy;
        const double k3vx = ax3, k3vy = ay3;

        const auto [ax4, ay4] = accel(x + dt * k3x, y + dt * k3y, cfg.gm);
        const double k4x = vx + dt * k3vx, k4y = vy + dt * k3vy;
        const double k4vx = ax4, k4vy = ay4;

        const double scale = dt / 6.0;
        x  += scale * (k1x  + 2.0 * k2x  + 2.0 * k3x  + k4x);
        y  += scale * (k1y  + 2.0 * k2y  + 2.0 * k3y  + k4y);
        vx += scale * (k1vx + 2.0 * k2vx + 2.0 * k3vx + k4vx);
        vy += scale * (k1vy + 2.0 * k2vy + 2.0 * k3vy + k4vy);
    }
};

int main() {
    constexpr SimConfig cfg{};
    OrbitalState state{};
    for (int i = 0; i < cfg.nsteps; ++i) {
        state.rk4_step(cfg);
    }
    std::cout << std::scientific << std::uppercase << std::setprecision(15)

4. Automated Verification Gate: verify_parity.pyScript wasit (referee) yang wajib dijalankan oleh agent untuk membuktikan hasil migrasi Rust & C++ identik dengan Fortran:Python#!/usr/bin/env python3
import subprocess
import sys

TOLERANCE = 1e-12


def run_cmd(cmd: list[str]) -> str:
  res = subprocess.run(cmd, capture_output=True, text=True, check=True)
  return res.stdout.strip()


def parse_vec(output: str) -> list[float]:
  return [float(x.replace("D", "E")) for x in output.split()]


def main():
  print("[1/3] Compiling Fortran, Rust, and C++20 binaries...")
  run_cmd(["gfortran", "-O2", "legacy_orbital.f", "-o", "bin_fortran"])
  run_cmd(["rustc", "-C", "opt-level=2", "orbital_solver.rs", "-o", "bin_rust"])
  run_cmd(
      ["g++", "-std=c++20", "-O2", "orbital_solver.cpp", "-o", "bin_cpp"]
  )

  print("[2/3] Executing simulations...")
  ref_f66 = parse_vec(run_cmd(["./bin_fortran"]))
  out_rust = parse_vec(run_cmd(["./bin_rust"]))
  out_cpp = parse_vec(run_cmd(["./bin_cpp"]))

  print(f"Fortran Ref : {ref_f66}")
  print(f"Rust Output : {out_rust}")
  print(f"C++20 Output: {out_cpp}")

  print("[3/3] Checking IEEE-754 numerical parity...")
  for idx, (f_val, r_val, c_val) in enumerate(zip(ref_f66, out_rust, out_cpp)):
    scale = max(1.0, abs(f_val))
    err_rust = abs(f_val - r_val) / scale
    err_cpp = abs(f_val - c_val) / scale
    if err_rust > TOLERANCE or err_cpp > TOLERANCE:
      print(
          f"FAIL at index {idx}: rel_err_rust={err_rust:.3e},"
          f" rel_err_cpp={err_cpp:.3e}"
      )
      sys.exit(1)

  print(
      f"PASS: Rust and C++20 match 1960s Fortran within relative tolerance"
      f" < {TOLERANCE}"
  )


if __name__ == "__main__":
  main()
