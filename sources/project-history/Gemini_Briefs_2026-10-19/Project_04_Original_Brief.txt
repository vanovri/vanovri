Langsung kita gas **Project #4: Desk Racer (04:20) — Dari Foto Meja Kerja Jadi Game Balap 3D Playable di Browser**.

Di demo ke-4 ini, Gemini 4 Argon melakukan orkestrasi lintas-modalitas yang keren banget:

1. **Multimodal Spatial Reconstruction:** Membaca foto meja kerja untuk memetakan posisi objek fisik (monitor, mechanical keyboard, cangkir kopi, tumpukan buku, mouse) ke koordinat bidang 3D $(X, Z)$ beserta *bounding box* kolisinya.
2. **Procedural 3D Asset & Track Generation:** Mengubah *scene graph* hasil ekstraksi foto menjadi *mesh* 3D rintangan meja dan *checkpoint* lintasan balap (via Blender MCP / Three.js).
3. **Playable Three.js Physics & Chase Cam Engine:** Merakit game balap RC di atas meja lengkap dengan fisika akselerasi/friksi/steering, deteksi tabrakan (*AABB & radial collision*), *chase camera* dinamis, dan *lap timer* *real-time*.

Di bawah ini adalah **4 komponen lengkap** untuk Project #4—termasuk file **`desk_racer.html` yang 100% langsung bisa lu buka dan mainkan di browser sekarang juga** pakai tombol `WASD` atau *Arrow Keys*:

### 1. Pipeline Vision-ke-3D Scene Graph (`desk_spatial_mapper.py`)

Script ini menerjemahkan deteksi objek 2D dari foto meja (koordinat piksel $[y_{\min}, x_{\min}, y_{\max}, x_{\max}]$ skala 0–1000 khas output *spatial bounding box* Gemini) menjadi layout dunia 3D (`desk_scene.json`) plus generator *waypoints* lintasan otomatis yang menghindari objek meja:

```python
#!/usr/bin/env python3
from dataclasses import asdict, dataclass
import json
import math
from typing import List


@dataclass
class DeskObstacle3D:
  id: str
  category: str
  x: float
  z: float
  width: float
  depth: float
  height: float
  color_hex: str


@dataclass
class TrackCheckpoint3D:
  index: int
  x: float
  z: float
  radius: float


def map_2d_photo_boxes_to_3d_desk(
    detections_2d: List[dict],
    desk_width_m: float = 160.0,
    desk_depth_m: float = 100.0,
) -> dict:
  """Mengonversi bounding box foto 2D [ymin, xmin, ymax, xmax] (skala 0..1000)

  menjadi koordinat meja 3D berpusat di (0, 0) untuk Three.js & Blender MCP.
  """
  height_catalog = {
      "monitor": (36.0, "#1e293b"),
      "keyboard": (3.2, "#334155"),
      "mug": (11.5, "#ef4444"),
      "books": (9.0, "#3b82f6"),
      "mouse": (3.5, "#64748b"),
      "lamp": (32.0, "#f59e0b"),
  }

  obstacles: List[DeskObstacle3D] = []
  for idx, det in enumerate(detections_2d):
    ymin, xmin, ymax, xmax = det["box_2d"]
    cat = det.get("label", "books").lower()

    # Normalisasi 0..1000 ke koordinat meja (-W/2 .. +W/2, -D/2 .. +D/2)
    cx_norm = ((xmin + xmax) / 2.0) / 1000.0 - 0.5
    cz_norm = ((ymin + ymax) / 2.0) / 1000.0 - 0.5

    w_world = max(6.0, ((xmax - xmin) / 1000.0) * desk_width_m)
    d_world = max(6.0, ((ymax - ymin) / 1000.0) * desk_depth_m)
    h_world, color = height_catalog.get(cat, (8.0, "#475569"))

    obstacles.append(
        DeskObstacle3D(
            id=f"{cat}_{idx}",
            category=cat,
            x=round(cx_norm * desk_width_m, 2),
            z=round(cz_norm * desk_depth_m, 2),
            width=round(w_world, 2),
            depth=round(d_world, 2),
            height=h_world,
            color_hex=color,
        )
    )

  # Buat 6 checkpoint melingkari objek di atas meja
  checkpoints: List[TrackCheckpoint3D] = []
  rx, rz = desk_width_m * 0.36, desk_depth_m * 0.34
  for i in range(6):
    angle = (2.0 * math.pi * i) / 6.0
    checkpoints.append(
        TrackCheckpoint3D(
            index=i,
            x=round(math.cos(angle) * rx, 2),
            z=round(math.sin(angle) * rz, 2),
            radius=11.0,
        )
    )

  return {
      "desk_dimensions": {"width": desk_width_m, "depth": desk_depth_m},
      "obstacles": [asdict(o) for o in obstacles],
      "checkpoints": [asdict(c) for c in checkpoints],
  }


if __name__ == "__main__":
  # Contoh hasil deteksi multimodal dari foto meja kerja
  sample_photo_detections = [
      {"label": "monitor", "box_2d": [80, 360, 230, 640]},
      {"label": "keyboard", "box_2d": [430, 370, 560, 610]},
      {"label": "mouse", "box_2d": [450, 660, 540, 720]},
      {"label": "mug", "box_2d": [260, 730, 350, 810]},
      {"label": "books", "box_2d": [220, 140, 380, 280]},
      {"label": "lamp", "box_2d": [100, 780, 210, 880]},
  ]

  scene_data = map_2d_photo_boxes_to_3d_desk(sample_photo_detections)
  with open("desk_scene.json", "w", encoding="utf-8") as f:
    json.dump(scene_data, f, indent=2)
  print(
      "[OK] Generated desk_scene.json with",
      len(scene_data["obstacles"]),
      "3D obstacles.",
  )

```

### 3. Automated Spatial & Track Referee (`verify_desk_racer.py`)

Di workflow Antigravity, kalau agent asal menaruh *checkpoint* di dalam *bounding box* monitor atau keyboard, mobil nggak bakal bisa menyelesaikan 1 lap. Script *referee* ini menguji `desk_scene.json` untuk memastikan seluruh *checkpoint* bebas dari kolisi objek meja dan lintasannya *solvable*:

```python
#!/usr/bin/env python3
import json
import math
import sys


def verify_scene_graph(json_path: str = "desk_scene.json") -> bool:
  with open(json_path, "r", encoding="utf-8") as f:
    scene = json.load(f)

  desk_w = scene["desk_dimensions"]["width"]
  desk_d = scene["desk_dimensions"]["depth"]
  obstacles = scene["obstacles"]
  checkpoints = scene["checkpoints"]

  print(
      f"[1/3] Validating {len(obstacles)} obstacles inside {desk_w}x{desk_d}"
      " desk bounds..."
  )
  for ob in obstacles:
    if abs(ob["x"]) + ob["width"] / 2 > desk_w / 2:
      print(f"FAIL: Obstacle {ob['id']} exceeds desk X boundary.")
      return False
    if abs(ob["z"]) + ob["depth"] / 2 > desk_d / 2:
      print(f"FAIL: Obstacle {ob['id']} exceeds desk Z boundary.")
      return False

  print(
      f"[2/3] Checking {len(checkpoints)} checkpoints for obstacle overlap..."
  )
  for cp in checkpoints:
    for ob in obstacles:
      dx = abs(cp["x"] - ob["x"])
      dz = abs(cp["z"] - ob["z"])
      if dx < (ob["width"] / 2 + 3.0) and dz < (ob["depth"] / 2 + 3.0):
        print(
            f"FAIL: Checkpoint #{cp['index']} collides with obstacle"
            f" '{ob['id']}'."
        )
        return False

  print("[3/3] Verifying closed-loop track perimeter length...")
  perimeter = 0.0
  for i in range(len(checkpoints)):
    c1 = checkpoints[i]
    c2 = checkpoints[(i + 1) % len(checkpoints)]
    seg_len = math.hypot(c2["x"] - c1["x"], c2["z"] - c1["z"])
    if seg_len < 15.0:
      print(
          f"FAIL: Checkpoints {c1['index']} and {c2['index']} are too close"
          f" ({seg_len:.1f}m)."
      )
      return False
    perimeter += seg_len

  print(
      "PASS: Desk Racer 3D scene graph is collision-free (Track length:"
      f" {perimeter:.1f} units)."
  )
  return True


if __name__ == "__main__":
  ok = verify_scene_graph()
  sys.exit(0 if ok else 1)
```
