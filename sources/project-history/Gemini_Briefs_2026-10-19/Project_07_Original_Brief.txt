Langsung kita sikat **Project #7: 3D DEXA Body Twin (07:35) — Transformasi Laporan Medis Datar Menjadi Interactive 3D Digital Body Composition Twin**.

Di demo ke-7 ini, Gemini 4 Argon menjembatani **ekstraksi dokumen klinis tak terstruktur** dengan **visualisasi anatomi 3D interaktif**:

1. **Clinical Telemetry Extraction & Conservation Gate:** Membaca laporan hasil pemindaian **DEXA (Dual-Energy X-ray Absorptiometry)** yang awalnya cuma tabel angka kaku per segmen tubuh (*Left Arm, Right Arm, Trunk, Pelvis, Left Leg, Right Leg, Head*), lalu mengekstrak *Tissue Fat %*, *Lean Muscle Mass (kg)*, *Bone Mineral Density (BMD $\text{g/cm}^2$)*, dan *Z-Score*.
2. **Physics/Mass Conservation Referee:** Sebelum data di-bind ke mesh 3D, agent menjalankan validasi hukum kekekalan massa—memastikan jumlah massa seluruh segmen anatomi sama dengan total berat badan pasien (toleransi deviasi di bawah 0.5%) dan menghitung indeks asimetri otot kiri-kanan (*Limb Symmetry Index*).
3. **Interactive 3D Anatomical Twin (Three.js + Raycasting):** Memetakan data klinis tadi ke bagian tubuh 3D yang bisa diputar, di-*hover*/klik per segmen (lengan, dada, panggul, kaki), lengkap dengan 3 mode *heatmap* (**Tissue Fat %**, **Lean Muscle Symmetry**, dan **X-Ray Skeletal BMD**).

Di bawah ini adalah **3 komponen lengkap** untuk Project #7—termasuk file **`dexa_body_twin.html` yang bisa langsung lu buka di browser sekarang** untuk menginspeksi *3D Digital Twin* secara *real-time*:

### 1. Parser Laporan DEXA & Referee Kekekalan Massa (`dexa_report_parser.py`)

Script ini mengekstrak teks laporan hasil scan DEXA mentah menjadi JSON terstruktur sekaligus bertindak sebagai **Referee Gate** yang memvalidasi kekekalan massa ($\sum \text{Massa Segmen} = \text{Total Massa}$) dan mendeteksi ketimpangan otot kiri-kanan:

```python
#!/usr/bin/env python3
from dataclasses import asdict, dataclass
import json
import re
import sys
from typing import Dict, List

RAW_DEXA_REPORT_TEXT = """
PATIENT ID: ANR-2026-DEXA | SCAN MODE: FULL BODY DUAL-ENERGY X-RAY
================================================================================
REGION       FAT(%)   FAT_G    LEAN_G    BMC_G    BMD_GCM2   Z_SCORE
--------------------------------------------------------------------------------
Head          21.4     1020      3450      510      2.18       +0.9
Left Arm      14.8      540      2920      195      0.96       +0.6
Right Arm     14.1      530      3040      205      0.99       +0.8
Trunk         16.2     4850     24100      890      1.12       +1.1
Pelvis        15.5     1620      8350      480      1.28       +1.3
Left Leg      15.9     1890      9420      580      1.36       +1.2
Right Leg     15.2     1840      9680      595      1.39       +1.4
================================================================================
TOTAL_REPORTED_MASS_KG: 74.88
"""

ROW_REGEX = re.compile(
    r"^(?P<region>Head|Left Arm|Right Arm|Trunk|Pelvis|Left Leg|Right Leg)\s+"
    r"(?P<fat_pct>\d+\.\d+)\s+"
    r"(?P<fat_g>\d+)\s+"
    r"(?P<lean_g>\d+)\s+"
    r"(?P<bmc_g>\d+)\s+"
    r"(?P<bmd>\d+\.\d+)\s+"
    r"(?P<z_score>[+\-]?\d+\.\d+)$",
    re.MULTILINE,
)


@dataclass
class AnatomicalRegionMetrics:
  region_id: str
  display_name: str
  fat_pct: float
  fat_kg: float
  lean_kg: float
  bmc_kg: float
  total_region_kg: float
  bmd_gcm2: float
  z_score: float


def parse_and_verify_dexa(raw_text: str) -> Dict:
  regions: Dict[str, AnatomicalRegionMetrics] = {}

  for match in ROW_REGEX.finditer(raw_text):
    name = match.group("region")
    reg_id = name.lower().replace(" ", "_")
    fat_kg = int(match.group("fat_g")) / 1000.0
    lean_kg = int(match.group("lean_g")) / 1000.0
    bmc_kg = int(match.group("bmc_g")) / 1000.0
    total_kg = round(fat_kg + lean_kg + bmc_kg, 3)

    regions[reg_id] = AnatomicalRegionMetrics(
        region_id=reg_id,
        display_name=name,
        fat_pct=float(match.group("fat_pct")),
        fat_kg=round(fat_kg, 3),
        lean_kg=round(lean_kg, 3),
        bmc_kg=round(bmc_kg, 3),
        total_region_kg=total_kg,
        bmd_gcm2=float(match.group("bmd")),
        z_score=float(match.group("z_score")),
    )

  # 1. Verifikasi Kelengkapan 7 Segmen Anatomi Wajib
  required_ids = {
      "head",
      "left_arm",
      "right_arm",
      "trunk",
      "pelvis",
      "left_leg",
      "right_leg",
  }
  missing = required_ids - set(regions.keys())
  if missing:
    raise ValueError(f"REFEREE GATE FAILED: Missing anatomical regions: {missing}")

  # 2. Verifikasi Kekekalan Massa (Sum of Segments vs Reported Total Mass)
  reported_match = re.search(r"TOTAL_REPORTED_MASS_KG:\s*(\d+\.\d+)", raw_text)
  reported_total_kg = float(reported_match.group(1))
  computed_total_kg = round(
      sum(r.total_region_kg for r in regions.values()), 2
  )
  mass_error_pct = (
      abs(computed_total_kg - reported_total_kg) / reported_total_kg * 100.0
  )

  if mass_error_pct > 0.5:
    raise ValueError(
        f"REFEREE GATE FAILED: Mass conservation error {mass_error_pct:.2f}% "
        f"(Computed={computed_total_kg}kg vs Reported={reported_total_kg}kg)"
    )

  # 3. Analisis Asimetri Otot Kiri-Kanan (Bilaterial Muscle Symmetry)
  arm_diff_pct = (
      (regions["right_arm"].lean_kg - regions["left_arm"].lean_kg)
      / regions["right_arm"].lean_kg
      * 100.0
  )
  leg_diff_pct = (
      (regions["right_leg"].lean_kg - regions["left_leg"].lean_kg)
      / regions["right_leg"].lean_kg
      * 100.0
  )

  total_fat_kg = round(sum(r.fat_kg for r in regions.values()), 2)
  total_lean_kg = round(sum(r.lean_kg for r in regions.values()), 2)
  total_bmc_kg = round(sum(r.bmc_kg for r in regions.values()), 2)

  return {
      "summary": {
          "total_mass_kg": computed_total_kg,
          "total_body_fat_pct": round(
              total_fat_kg / computed_total_kg * 100.0, 1
          ),
          "total_lean_kg": total_lean_kg,
          "total_fat_kg": total_fat_kg,
          "total_bmc_kg": total_bmc_kg,
          "arm_lean_asymmetry_pct": round(arm_diff_pct, 2),
          "leg_lean_asymmetry_pct": round(leg_diff_pct, 2),
          "mass_conservation_error_pct": round(mass_error_pct, 4),
      },
      "regions": {k: asdict(v) for k, v in regions.items()},
  }


if __name__ == "__main__":
  twin_payload = parse_and_verify_dexa(RAW_DEXA_REPORT_TEXT)
  with open("dexa_twin_data.json", "w", encoding="utf-8") as f:
    json.dump(twin_payload, f, indent=2)

  s = twin_payload["summary"]
  print(
      f"[PASS] 7 Anatomical Regions Extracted | Total Mass:"
      f" {s['total_mass_kg']} kg (Error: {s['mass_conservation_error_pct']}%)"
  )
  print(
      f"       Body Fat: {s['total_body_fat_pct']}% | Lean Mass:"
      f" {s['total_lean_kg']} kg | Bone Mineral: {s['total_bmc_kg']} kg"
  )
  print(
      f"       Limb Asymmetry -> Arms: {s['arm_lean_asymmetry_pct']:+.2f}% |"
      f" Legs: {s['leg_lean_asymmetry_pct']:+.2f}%"
  )

```
