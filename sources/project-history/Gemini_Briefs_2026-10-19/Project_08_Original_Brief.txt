Di demo ke-8 ini, tantangan utamanya bukan sekadar menggambar bola Bumi dan Bulan di layar, melainkan **ketepatan matematika mekanika orbit dan parsing telemetri standar penerbangan antariksa**:

1. **Raw CCSDS OEM Telemetry Ingestion:** Membaca file **CCSDS Orbit Ephemeris Message (OEM 2.0)**—format standar NASA yang berisi metadata kerangka acuan (`EME2000` / Earth-Centered Inertial) dan baris *state vector* 6-DOF berstempel waktu: posisi $(X, Y, Z)$ dalam $\text{km}$ dan kecepatan $(V_x, V_y, V_z)$ dalam $\text{km/s}$.
2. **Astrodynamics & Physics Referee Gate:** Sebelum di-render ke 3D, agent memverifikasi hukum fisika astrodinamika—menghitung jarak ke pusat Bumi ($r_E$), jarak ke Bulan ($r_M$), kecepatan heliocentric/geocentric ($\vert{}v\vert{}$), dan **Specific Orbital Energy** ($\epsilon = \frac{v^2}{2} - \frac{\mu_E}{r_E} - \frac{\mu_M}{r_M}$), serta memastikan lintasan *free-return figure-8* tidak menabrak permukaan Bumi ($R_E = 6378.1\text{ km}$) atau Bulan ($R_M = 1737.4\text{ km}$).
3. **Real-Time 3D Cislunar Mission Control (`artemis2_visualizer.html`):** Mengonversi koordinat ratusan ribu kilometer ke ruang WebGL/Three.js tanpa *floating-point jitter*, lengkap dengan *Mission Elapsed Time (MET) Scrubber*, vektor kecepatan 3D, dan pengunci kamera (*Earth Lock*, *Orion Spacecraft Chase*, *Lunar Flyby Lock*).

### 1. NASA CCSDS OEM Parser & Astrodynamics Referee (`artemis2_ephemeris_parser.py`)

Script ini mem-parsing stream telemetri **NASA CCSDS OEM 2.0**, menghitung parameter mekanika orbit di setiap titik waktu, dan menjalankan **Physics Referee Gate** sebelum mengekspor `artemis2_trajectory.json`:

```python
#!/usr/bin/env python3
from dataclasses import asdict, dataclass
import json
import math
import sys
from typing import List

# Konstanta Astrodinamika Standar (WGS84 / JPL DE440)
MU_EARTH_KM3_S2 = 398600.4418
MU_MOON_KM3_S2 = 4902.800066
RADIUS_EARTH_KM = 6378.137
RADIUS_MOON_KM = 1737.400

# Sampel Telemetri NASA CCSDS OEM 2.0 (Artemis II Free-Return Cislunar Profile)
# Format kolom data: EPOCH_ISO  X(km)  Y(km)  Z(km)  VX(km/s)  VY(km/s)  VZ(km/s)
RAW_CCSDS_OEM_STREAM = """CCSDS_OEM_VERS = 2.0
CREATION_DATE  = 2026-04-01T00:00:00.000
ORIGINATOR     = NASA_JSC_FD_ORION

META_START
OBJECT_NAME          = ORION_ARTEMIS_II
OBJECT_ID            = 2026-001A
CENTER_NAME          = EARTH
REF_FRAME            = EME2000
TIME_SYSTEM          = UTC
META_STOP

2026-04-02T00:00:00.000    6563.0      0.0       0.0      0.000   10.840   0.950
2026-04-03T00:00:00.000  -88420.0  112500.0   11200.0     -1.320    1.890   0.180
2026-04-04T00:00:00.000 -184500.0  214800.0   21900.0     -0.980    1.210   0.110
2026-04-05T00:00:00.000 -269000.0  292400.0   28400.0     -0.780    0.790   0.040
2026-04-06T06:00:00.000 -317200.0  243900.0   22100.0      0.920   -1.180  -0.210
2026-04-07T00:00:00.000 -258400.0  152600.0   14200.0      0.860   -1.050  -0.120
2026-04-08T00:00:00.000 -174200.0   88400.0    8100.0      1.150   -1.340  -0.110
2026-04-09T00:00:00.000  -79500.0   32100.0    2900.0      1.980   -2.420  -0.190
2026-04-10T00:00:00.000    6510.0   -1240.0    -180.0      1.850  -10.790  -0.820
"""


@dataclass
class EphemerisStateVector:
  epoch: str
  x_km: float
  y_km: float
  z_km: float
  vx_kms: float
  vy_kms: float
  vz_kms: float
  earth_alt_km: float
  lunar_alt_km: float
  speed_kms: float
  specific_energy_km2_s2: float


def approximate_moon_position_km(step_idx: int, total_steps: int) -> tuple[float, float, float]:
  """Menghitung posisi orbit Bulan pada kerangka EME2000 selama jendela misi."""
  lunar_dist = 384400.0
  # Saat step ke-4 (2026-04-06T06:00), Orion berada di titik perilselene (flyby belakang Bulan)
  base_angle = 2.48
  angle_offset = (step_idx - 4) * 0.21
  theta = base_angle + angle_offset
  mx = lunar_dist * math.cos(theta)
  my = lunar_dist * math.sin(theta)
  mz = 21500.0 * math.sin(theta)
  return mx, my, mz


def parse_and_verify_oem(raw_oem: str) -> dict:
  in_meta = False
  meta = {}
  raw_rows = []

  for line in raw_oem.splitlines():
    stripped = line.strip()
    if not stripped or stripped.startswith("COMMENT"):
      continue
    if stripped == "META_START":
      in_meta = True
      continue
    if stripped == "META_STOP":
      in_meta = False
      continue
    if in_meta or "=" in stripped:
      k, v = [p.strip() for p in stripped.split("=", 1)]
      meta[k] = v
      continue

    parts = stripped.split()
    if len(parts) == 7:
      raw_rows.append(parts)

  if meta.get("REF_FRAME") != "EME2000":
    raise ValueError(f"Unsupported reference frame: {meta.get('REF_FRAME')}")

  states: List[EphemerisStateVector] = []
  min_earth_alt = float("inf")
  min_lunar_alt = float("inf")

  for idx, parts in enumerate(raw_rows):
    epoch = parts[0]
    x, y, z = float(parts[1]), float(parts[2]), float(parts[3])
    vx, vy, vz = float(parts[4]), float(parts[5]), float(parts[6])

    r_earth = math.sqrt(x * x + y * y + z * z)
    earth_alt = r_earth - RADIUS_EARTH_KM

    mx, my, mz = approximate_moon_position_km(idx, len(raw_rows))
    r_moon = math.sqrt((x - mx) ** 2 + (y - my) ** 2 + (z - mz) ** 2)
    lunar_alt = r_moon - RADIUS_MOON_KM

    speed = math.sqrt(vx * vx + vy * vy + vz * vz)
    spec_energy = (
        0.5 * (speed ** 2)
        - (MU_EARTH_KM3_S2 / r_earth)
        - (MU_MOON_KM3_S2 / max(r_moon, 1.0))
    )

    min_earth_alt = min(min_earth_alt, earth_alt)
    min_lunar_alt = min(min_lunar_alt, lunar_alt)

    states.append(
        EphemerisStateVector(
            epoch=epoch,
            x_km=x,
            y_km=y,
            z_km=z,
            vx_kms=vx,
            vy_kms=vy,
            vz_kms=vz,
            earth_alt_km=round(earth_alt, 2),
            lunar_alt_km=round(lunar_alt, 2),
            speed_kms=round(speed, 4),
            specific_energy_km2_s2=round(spec_energy, 4),
        )
    )

  # --- ASTRODYNAMICS REFEREE GATES ---
  if min_earth_alt <= 120.0:
    raise ValueError(
        f"REFEREE GATE FAILED: Trajectory intersects Earth surface/atmosphere"
        f" (min_earth_alt={min_earth_alt:.1f} km)"
    )
  if min_lunar_alt <= 100.0:
    raise ValueError(
        f"REFEREE GATE FAILED: Lunar impact detected"
        f" (min_lunar_alt={min_lunar_alt:.1f} km)"
    )
  if not (9.5 <= states[0].speed_kms <= 11.5):
    raise ValueError(
        f"REFEREE GATE FAILED: Invalid Trans-Lunar Injection (TLI) velocity:"
        f" {states[0].speed_kms} km/s"
    )

  return {
      "metadata": meta,
      "summary": {
          "vector_count": len(states),
          "tli_injection_speed_kms": states[0].speed_kms,
          "closest_lunar_flyby_alt_km": round(min_lunar_alt, 1),
          "max_earth_distance_km": round(
              max(
                  math.hypot(s.x_km, s.y_km, s.z_km) for s in states
              ),
              1,
          ),
      },
      "state_vectors": [asdict(s) for s in states],
  }


if __name__ == "__main__":
  payload = parse_and_verify_oem(RAW_CCSDS_OEM_STREAM)
  with open("artemis2_trajectory.json", "w", encoding="utf-8") as f:
    json.dump(payload, f, indent=2)

  s = payload["summary"]
  print(
      f"[PASS] Parsed {s['vector_count']} CCSDS OEM state vectors"
      f" (Frame: {payload['metadata']['REF_FRAME']})"
  )
  print(
      f"       TLI Speed: {s['tli_injection_speed_kms']} km/s | Max Apogee:"
      f" {s['max_earth_distance_km']} km | Lunar Flyby Alt:"
      f" {s['closest_lunar_flyby_alt_km']} km"
  )

```

      const specEnergy = 0.5 * speedKms * speedKms - (MU_EARTH / rEarthKm) - (MU_MOON / rMoonKm);

      document.getElementById("val-earth-dist").textContent = `${Math.round(rEarthKm).toLocaleString()} km`;

      document.getElementById("val-xyz").textContent =
        `[ ${Math.round(xKm).toLocaleString()}, ${Math.round(yKm).toLocaleString()}, ${Math.round(zKm).toLocaleString()} ]`;

      // Mission Elapsed Time (10 days total = 240 hours)
      const totalHours = t * 240.0;
      const days = Math.floor(totalHours / 24);
      const hours = Math.floor(totalHours % 24);
      const mins = Math.floor((totalHours * 60) % 60);
      const isoDay = String(2 + days).padStart(2, "0");
      const isoHr = String(hours).padStart(2, "0");
      const isoMin = String(mins).padStart(2, "0");

      document.getElementById("met-clock").textContent =
        `MET: T+ 0${days}d ${isoHr}h ${isoMin}m | Epoch: 2026-04-${isoDay}T${isoHr}:${isoMin}:00Z`;

      const phaseEl = document.getElementById("mission-phase");
      if (t < 0.08) phaseEl.textContent = "Phase 1: Trans-Lunar Injection (TLI Burn)";
      else if (t < 0.44) phaseEl.textContent = "Phase 2: Outbound Cislunar Coast";
      else if (t < 0.58) phaseEl.textContent = "Phase 3: Lunar Perilselene Flyby (Closest Approach)";
      else if (t < 0.93) phaseEl.textContent = "Phase 4: Trans-Earth Free-Return Coast";
      else phaseEl.textContent = "Phase 5: Earth Entry Interface & Splashdown";
    }

    // Orbit Drag Camera Support
    let camYaw = 0.5, camPitch = 0.55, camDist = 115;
    let isDragging = false, prevX = 0, prevY = 0;
    window.addEventListener("mousedown", e => {
      if (e.target.tagName === "INPUT" || e.target.tagName === "BUTTON") return;
      isDragging = true; prevX = e.clientX; prevY = e.clientY;
    });
    window.addEventListener("mouseup", () => isDragging = false);
    window.addEventListener("mousemove", e => {
      if (!isDragging) return;
      camYaw -= (e.clientX - prevX) * 0.007;
      camPitch = Math.max(0.1, Math.min(1.45, camPitch + (e.clientY - prevY) * 0.006));
      prevX = e.clientX; prevY = e.clientY;
    });
    window.addEventListener("wheel", e => {
      camDist = Math.max(18, Math.min(220, camDist + e.deltaY * 0.05));
    });

    function animate() {
      requestAnimationFrame(animate);

      if (isPlaying) {
        missionT += 0.00085;
        if (missionT > 1.0) missionT = 0.0;
        slider.value = Math.round(missionT * 1000);

      const orionPos = trajectoryCurve.getPointAt(missionT);
      const tangent = trajectoryCurve.getTangentAt(missionT).normalize();

      orionGroup.position.copy(orionPos);
      orionGroup.lookAt(orionPos.clone().add(tangent));

      velArrow.position.copy(orionPos);
      velArrow.setDirection(tangent);

      moonMesh.position.copy(moonPos);

      updateTelemetryHUD(orionPos, tangent, moonPos, missionT);

      // Camera Mode Routing
      let focusTarget = new THREE.Vector3(-25, 0, -25);
      let activeDist = camDist;
      if (camMode === "orion") {
        focusTarget.copy(orionPos);
        activeDist = 22;
      } else if (camMode === "moon") {
        focusTarget.copy(moonPos);
        activeDist = 26;
      }

      camera.position.set(
        focusTarget.x + Math.sin(camYaw) * Math.cos(camPitch) * activeDist,
        focusTarget.y + Math.sin(camPitch) * activeDist,
        focusTarget.z + Math.cos(camYaw) * Math.cos(camPitch) * activeDist
      );
      camera.lookAt(focusTarget);

      renderer.render(scene, camera);
    }
    animate();

    window.addEventListener("resize", () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
